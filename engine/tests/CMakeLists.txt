# Linux/CI unit test for the engine core (no GUI, no audio device)
cmake_minimum_required(VERSION 3.22)
project(JAJAEngineTest VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 20)
if(NOT JUCE_DIR)
  include(FetchContent)
  FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(juce)
else()
  add_subdirectory(${JUCE_DIR} juce)
endif()
juce_add_console_app(EngineTest PRODUCT_NAME "EngineTest")
target_sources(EngineTest PRIVATE EngineTest.cpp ../src/Engine.cpp)
target_include_directories(EngineTest PRIVATE ../src)
target_compile_definitions(EngineTest PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=0 JUCE_PLUGINHOST_LV2=0 JUCE_PLUGINHOST_LADSPA=0 JUCE_STANDALONE_APPLICATION=1)
target_link_libraries(EngineTest PRIVATE juce::juce_core juce::juce_events juce::juce_audio_basics juce::juce_audio_formats juce::juce_audio_processors_headless juce::juce_recommended_config_flags)
